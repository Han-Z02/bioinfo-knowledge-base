贝叶斯优化目标函数可以自己定义如下

def xgb_cv(max_depth, learning_rate, reg_lambda, subsample):    """贝叶斯优化目标函数：最大化负RMSE"""    max_depth = int(max_depth)    model = XGBRegressor(        max_depth=max_depth,        learning_rate=learning_rate,        reg_lambda=reg_lambda,        subsample=subsample,        objective='reg:squarederror',        n_estimators=5000,        random_state=42,        tree_method='hist',        eval_metric='rmse'    )    model.fit(        X_train_nor, y_train_nor,        eval_set=[(X_vaild_nor, y_vaild_nor)],        verbose=False,        early_stopping_rounds=100    )    pred = model.predict(X_vaild_nor)    rmse = root_mean_squared_error(y_vaild_nor, pred)    return -rmse
